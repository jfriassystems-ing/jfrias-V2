// home.js - OPTIMIZADO document.addEventListener('DOMContentLoaded', () => { // Animación de entrada con IntersectionObserver (más eficiente) const techCards = document.querySelectorAll('.tech-card'); if ('IntersectionObserver' in window && techCards.length > 0) { const observer = new IntersectionObserver((entries) => { entries.forEach((entry, index) => { if (entry.isIntersecting) { const card = entry.target; setTimeout(() => { card.style.transition = 'all 0.5s ease'; card.style.opacity = '1'; card.style.transform = 'translateY(0)'; }, index * 100); observer.unobserve(card); } }); }, { threshold: 0.1 }); techCards.forEach(card => { card.style.opacity = '0'; card.style.transform = 'translateY(20px)'; observer.observe(card); }); } else { // Fallback para navegadores antiguos techCards.forEach(card => { card.style.opacity = '1'; card.style.transform = 'translateY(0)'; }); } console.log('Home page loaded'); });
